    function setLanguage(lang) {
      const t = translations[lang];
      document.getElementById('html-root').dir = t.dir;
      document.getElementById('ui-subtitle').innerHTML = t.subtitle;
      document.getElementById('upload-title').innerHTML = t.uploadTitle;
      document.getElementById('upload-desc').innerHTML = t.uploadDesc;
      document.getElementById('btn-choose-file').innerText = t.chooseFile;
      
      const btnAdd = document.getElementById('btn-add-text');
      if (btnAdd) btnAdd.innerText = t.addText;
      
      const btnDl = document.getElementById('btn-download');
      if (btnDl) btnDl.innerText = t.download;

      const lblPg = document.getElementById('lbl-page');
      if (lblPg) lblPg.innerText = t.page;

      document.getElementById('btn-donate-text').innerText = t.donate;
      document.getElementById('btn-donate-main').innerText = t.donate;

    function loadDocument(e) {
      const file = e.target.files[0];
      if (!file) return;

      document.getElementById('upload-box').classList.add('hidden');
      document.getElementById('editor-wrapper').classList.remove('hidden');
      document.getElementById('editor-controls').classList.remove('hidden');

      const reader = new FileReader();

      if (file.type === 'application/pdf') {
        reader.onload = function(evt) {
          const typedarray = new Uint8Array(evt.target.result);
          pdfjsLib.getDocument(typedarray).promise.then(pdf => {
            pdfDoc = pdf;
            document.getElementById('page-count').innerText = pdf.numPages;
            renderPage(pageNum);
          });
        };
        reader.readAsArrayBuffer(file);
      } else {
        reader.onload = function(evt) {
          const img = new Image();
          img.onload = function() {
            const maxWidth = 800;
            const ratio = maxWidth / img.width;
            canvas.width = maxWidth;
            canvas.height = img.height * ratio;

            inputsLayer.style.width = canvas.width + 'px';
            inputsLayer.style.height = canvas.height + 'px';

            ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
          };
          img.src = evt.target.result;
        };
        reader.readAsDataURL(file);
      }
    }

    function renderPage(num) {
      pdfDoc.getPage(num).then(page => {
        const viewport = page.getViewport({ scale: 1.5 });
        canvas.width = viewport.width;
        canvas.height = viewport.height;

        inputsLayer.style.width = viewport.width + 'px';
        inputsLayer.style.height = viewport.height + 'px';

        page.render({ canvasContext: ctx, viewport: viewport });
        document.getElementById('page-num').innerText = num;
      });
    }

    function handleCanvasClick(e) {
      if (e.target.tagName === 'INPUT') return;

      const rect = document.getElementById('canvas-wrapper').getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;

      createTextInput(x, y);
    }

    function addTextAtCenter() {
      createTextInput(100, 100);
    }

    function createTextInput(x, y) {
      const input = document.createElement('input');
      input.type = 'text';
      input.className = 'global-text-input pointer-events-auto';
      input.placeholder = 'Type / הקלד / اكتب...';
      input.dir = 'auto';

      const fontSize = document.getElementById('text-size').value;
      const color = document.getElementById('text-color').value;

      input.style.fontSize = fontSize + 'px';
      input.style.color = color;
      input.style.left = (x - 10) + 'px';
      input.style.top = (y - 12) + 'px';
      input.style.minWidth = '160px';

      makeDraggable(input);
      inputsLayer.appendChild(input);
      setTimeout(() => input.focus(), 50);
    }

    function makeDraggable(element) {
      let posX = 0, posY = 0, mouseX = 0, mouseY = 0;

      element.onmousedown = function(e) {
        if (document.activeElement === element) return;

        e.preventDefault();
        mouseX = e.clientX;
        mouseY = e.clientY;

        document.onmouseup = function() {
          document.onmousemove = null;
          document.onmouseup = null;
        };

        document.onmousemove = function(e) {
          e.preventDefault();
          posX = mouseX - e.clientX;
          posY = mouseY - e.clientY;
          mouseX = e.clientX;
          mouseY = e.clientY;
          element.style.top = (element.offsetTop - posY) + "px";
          element.style.left = (element.offsetLeft - posX) + "px";
        };
      };
    }

    function prevPage() {
      if (pageNum <= 1) return;
      pageNum--;
      renderPage(pageNum);
    }

    function nextPage() {
      if (!pdfDoc || pageNum >= pdfDoc.numPages) return;
      pageNum++;
      renderPage(pageNum);
    }

    function downloadPDF() {
      const { jsPDF } = window.jspdf;

      const tempCanvas = document.createElement('canvas');
      tempCanvas.width = canvas.width;
      tempCanvas.height = canvas.height;
      const tempCtx = tempCanvas.getContext('2d');

      tempCtx.drawImage(canvas, 0, 0);

      const inputs = document.querySelectorAll('.global-text-input');
      inputs.forEach(inp => {
        if (inp.value.trim() !== '') {
          const fontSize = parseInt(inp.style.fontSize) || 16;
          const x = parseInt(inp.style.left) + 6;
          const y = parseInt(inp.style.top) + (fontSize * 1.1);

          tempCtx.font = `bold ${fontSize}px Tajawal, Rubik, Inter, sans-serif`;
          tempCtx.fillStyle = inp.style.color || "#000000";
          tempCtx.fillText(inp.value, x, y);
        }
      });

      const imgData = tempCanvas.toDataURL('image/jpeg', 1.0);
      const pdf = new jsPDF({
        orientation: tempCanvas.width > tempCanvas.height ? 'landscape' : 'portrait',
        unit: 'px',
        format: [tempCanvas.width, tempCanvas.height]
      });

      pdf.addImage(imgData, 'JPEG', 0, 0, tempCanvas.width, tempCanvas.height);
      pdf.save('heib12_filled_document.pdf');
    }
